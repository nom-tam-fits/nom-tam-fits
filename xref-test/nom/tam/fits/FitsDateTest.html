48      @Test
49      public void testIsoDateParsing() throws Exception {
50          Assertions.assertEquals(REF_TIME_MS, new FitsDate("2018-11-28T12:13:14").toDate().getTime());
51          Assertions.assertEquals(REF_TIME_MS + 100, new FitsDate("2018-11-28T12:13:14.1").toDate().getTime());
52          Assertions.assertEquals(REF_TIME_MS + 120, new FitsDate("2018-11-28T12:13:14.12").toDate().getTime());
53          Assertions.assertEquals(REF_TIME_MS + 123, new FitsDate("2018-11-28T12:13:14.123").toDate().getTime());
54          Assertions.assertEquals(REF_TIME_MS + 123, new FitsDate("2018-11-28T12:13:14.1234").toDate().getTime());
55          Assertions.assertEquals(REF_TIME_MS + 124, new FitsDate("2018-11-28T12:13:14.1236").toDate().getTime());
56          Assertions.assertEquals(REF_TIME_MS + 123, new FitsDate("2018-11-28T12:13:14.12345").toDate().getTime());
57          Assertions.assertEquals(REF_TIME_MS + 124, new FitsDate("2018-11-28T12:13:14.123567").toDate().getTime());
58          Assertions.assertEquals(REF_TIME_MS + 10, new FitsDate("2018-11-28T12:13:14.01").toDate().getTime());
59          Assertions.assertEquals(REF_TIME_MS + 1, new FitsDate("2018-11-28T12:13:14.001").toDate().getTime());
60          Assertions.assertEquals(REF_TIME_MS, new FitsDate("2018-11-28T12:13:14.0001").toDate().getTime());
61      }

121     @Test
122     public void testGetFitsDateStringRangeValidation() {
123         Calendar cal = new GregorianCalendar(TimeZone.getTimeZone("UTC"));
124         cal.set(Calendar.ERA, GregorianCalendar.AD);
125         cal.set(Calendar.YEAR, 99999);
126         cal.set(Calendar.MONTH, Calendar.JANUARY);
127         cal.set(Calendar.DAY_OF_MONTH, 1);
128         Assertions.assertEquals("+99999-01-01", FitsDate.getFitsDateString(cal.getTime(), false));
129 
130         cal.set(Calendar.YEAR, 100000);
131         Assertions.assertThrows(FitsException.class, () -> FitsDate.getFitsDateString(cal.getTime(), false));
132 
133         cal.set(Calendar.ERA, GregorianCalendar.BC);
134         cal.set(Calendar.YEAR, 100000); // Calendar YEAR 100000 BC == FITS year -99999
135         Assertions.assertEquals("-99999-01-01", FitsDate.getFitsDateString(cal.getTime(), false));
136 
137         cal.set(Calendar.YEAR, 100001); // Calendar YEAR 100001 BC == FITS year -100000
138         Assertions.assertThrows(FitsException.class, () -> FitsDate.getFitsDateString(cal.getTime(), false));
139     }

141     @Test
142     public void testCalendarEraMapping() throws Exception {
143         assertEraYear(new FitsDate("0001-06-15"), GregorianCalendar.AD, 1);
144         assertEraYear(new FitsDate("0000-06-15"), GregorianCalendar.BC, 1);
145         assertEraYear(new FitsDate("-00001-06-15"), GregorianCalendar.BC, 2);
146         assertEraYear(new FitsDate("-00999-06-15"), GregorianCalendar.BC, 1000);
147         assertEraYear(new FitsDate("-99999-06-15"), GregorianCalendar.BC, 100000);
148     }

150     private static void assertEraYear(FitsDate fitsDate, int expectedEra, int expectedYear) {
151         Calendar cal = new GregorianCalendar(TimeZone.getTimeZone("UTC"));
152         cal.setTime(fitsDate.toDate());
153         Assertions.assertEquals(expectedEra, cal.get(Calendar.ERA));
154         Assertions.assertEquals(expectedYear, cal.get(Calendar.YEAR));
155     }

194     @Test
195     public void testToStringAndToDateAgreeOnLenientCalendarNormalization() throws Exception {
196         FitsDate fitsDate = new FitsDate("2023-02-30");
197         Calendar cal = new GregorianCalendar(TimeZone.getTimeZone("UTC"));
198         cal.setTime(fitsDate.toDate());
199         int rolledOverMonth = cal.get(Calendar.MONTH) + 1;
200         int rolledOverDay = cal.get(Calendar.DAY_OF_MONTH);
201         String expected = String.format("2023-%02d-%02d", rolledOverMonth, rolledOverDay);
202         Assertions.assertEquals(expected, fitsDate.toString());
203     }
